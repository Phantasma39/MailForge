# MailServer 的 POP3 用户账号文件
# ============================================================
# 格式：一行一个账号：用户名:密码
#   - "#" 开头的行是注释，会被忽略
#   - 用户名会自动"转小写并去掉 @ 后面的域名"再参与匹配，
#     所以写 bob、Bob、bob@example.com 效果都一样
#   - 用户名对应的收件目录是 ./mailbox/<用户名>/，
#     例如 bob 的邮件在 ./mailbox/bob/ 下面
#   - 想新增用户：在下面加一行，并等 SMTP 服务器第一次给 TA 投递邮件时，
#     服务器会自动创建 ./mailbox/<用户名>/ 目录
# ============================================================

bob:123456
alice:123456
phant:ra4s888
